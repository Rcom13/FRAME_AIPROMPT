OUTDIR=~/aws_inv_$(date +%Y%m%d_%H%M); mkdir -p $OUTDIR; r=ap-northeast-1
aws resourcegroupstaggingapi get-resources --region $r --output json > $OUTDIR/tagged.json 2>/dev/null
aws ec2 describe-vpcs --region $r --output json > $OUTDIR/vpcs.json 2>/dev/null
aws ec2 describe-subnets --region $r --output json > $OUTDIR/subnets.json 2>/dev/null
aws ec2 describe-route-tables --region $r --output json > $OUTDIR/rtb.json 2>/dev/null
aws ec2 describe-security-groups --region $r --output json > $OUTDIR/sg.json 2>/dev/null
aws ec2 describe-network-interfaces --region $r --output json > $OUTDIR/eni.json 2>/dev/null
aws ec2 describe-instances --region $r --output json > $OUTDIR/ec2.json 2>/dev/null
aws elbv2 describe-load-balancers --region $r --output json > $OUTDIR/elb.json 2>/dev/null
aws elbv2 describe-target-groups --region $r --output json > $OUTDIR/tg.json 2>/dev/null
aws rds describe-db-instances --region $r --output json > $OUTDIR/rds.json 2>/dev/null
aws lambda list-functions --region $r --output json > $OUTDIR/lambda.json 2>/dev/null
aws configservice select-resource-config --expression "SELECT resourceId, resourceType, resourceName, relationships, tags" --region $r --output json > $OUTDIR/config.json 2>/dev/null
aws s3api list-buckets --output json > $OUTDIR/s3.json 2>/dev/null
aws iam list-roles --output json > $OUTDIR/iam_roles.json 2>/dev/null
aws route53 list-hosted-zones --output json > $OUTDIR/route53.json 2>/dev/null
aws cloudfront list-distributions --output json > $OUTDIR/cloudfront.json 2>/dev/null
find $OUTDIR -size -5c -delete
zip -rq $OUTDIR.zip $OUTDIR && echo "完成，下载路径: $OUTDIR.zip"
